% Figure 1: the verifier-gap instrument.
% Blue = exact truth, vermillion = degraded proxy, green = what the instrument outputs.
% Grouped-panel layout. Every connector is routed orthogonally; no diagonals.
% Coordinates are explicit (cm) so the routing stays stable when content is edited.
% Kept narrow on purpose: the picture is \resizebox'd, so extra natural width
% shrinks every label. Measure before widening.
\begin{tikzpicture}[
  font=\small,
  x=1cm, y=1cm,
  ttl/.style={font=\scriptsize\itshape, inner sep=0pt, anchor=north west},
  bx/.style={draw, rounded corners=2pt, fill=white, align=center,
             inner sep=3pt, line width=0.45pt, font=\scriptsize},
  btruth/.style={bx, draw=blue!45!black},
  bout/.style={bx, draw=green!40!black},
  bneut/.style={bx, draw=black!50},
  chip/.style={draw, rounded corners=2pt, fill=white, draw=red!50!black,
               align=center, inner sep=2.4pt, line width=0.45pt, font=\scriptsize},
  arr/.style={-{Stealth[length=1.9mm]}, line width=0.6pt, black!65},
  tarr/.style={arr, draw=blue!50!black},
  parr/.style={arr, draw=red!55!black},
  oarr/.style={arr, draw=green!40!black},
  garr/.style={arr, draw=green!40!black, dashed, dash pattern=on 2pt off 1.6pt},
  lab/.style={font=\scriptsize\itshape, fill=white, inner sep=1.2pt},
  labA/.style={lab, fill=blue!5},   % sits inside panel A, must match its fill
  hop/.style={preaction={draw=white, line width=2.4pt, -}},
]

% =====================  PANEL A : the exact oracle  =====================
\fill[blue!5, rounded corners=3pt] (0,5.62) rectangle (14,7.52);

  {Decision tensor $T$\\[-1.5pt]$\NContexts$ contexts\\[-1.5pt]every reason $\mathrm{MSR}(c)$};

\fill[black!3, rounded corners=3pt] (0,0.10) rectangle (14,2.10);
